const API_BASE_URL = "http://localhost:8080/api/v1";


/**
 * Gá»i API chung.
 *
 * JWT náº±m trong HttpOnly Cookie "accessToken",
 * vÃ¬ váº­y frontend pháº£i dÃ¹ng:
 *
 * credentials: "include"
 */
async function apiRequest(
    endpoint,
    options = {}
) {

    const config = {
        ...options,

        credentials: "include",

        headers: {
            "Content-Type": "application/json",

            ...(options.headers || {})
        }
    };


    let response;

    try {

        response = await fetch(
            API_BASE_URL + endpoint,
            config
        );

    } catch (error) {

        console.error(
            "API CONNECTION ERROR:",
            error
        );

        throw new Error(
            "KhÃ´ng thá»ƒ káº¿t ná»‘i tá»›i mÃ¡y chá»§."
        );
    }


    // =================================================
    // HTTP 401
    // =================================================

    if (response.status === 401) {

        console.error(
            "API 401 - Unauthorized"
        );

        sessionStorage.clear();

        window.location.href =
            "../login/login.html";

        throw new Error(
            "PhiÃªn Ä‘Äƒng nháº­p Ä‘Ã£ háº¿t háº¡n."
        );
    }


    // =================================================
    // HTTP 204
    // =================================================

    if (response.status === 204) {

        return null;
    }


    // =================================================
    // Äá»c JSON
    // =================================================

    let result;

    try {

        result = await response.json();

    } catch (error) {

        console.error(
            "API RESPONSE IS NOT JSON:",
            error
        );

        throw new Error(
            "Server tráº£ vá» dá»¯ liá»‡u khÃ´ng há»£p lá»‡."
        );
    }


    // =================================================
    // HTTP ERROR
    // =================================================

    if (!response.ok) {

        throw new Error(
            result.message ||
            "API request tháº¥t báº¡i."
        );
    }


    // =================================================
    // ApiResponse cá»§a Backend
    // =================================================

    if (
        result &&
        result.code &&
        result.code !== "success"
    ) {

        throw new Error(
            result.message ||
            "API request tháº¥t báº¡i."
        );
    }


    return result;
}
